import {SWITCH_THEME} from './actionTypes';

const initialState = {
  theme: 'light',
};

const ThemeSwitchwerReducer = (state = initialState, action) => {
  //console.log( action.type, ' , - ',action.payload , ' , ,aaa ThemeSwitchwerReducer' );
  switch (action.type) {
    case SWITCH_THEME:
      return {
        ...state,
        theme: action.payload.theme,
      };
    default:
      return state;
  }
};

export default ThemeSwitchwerReducer;
