import AsyncStorage from '@react-native-async-storage/async-storage';

export const PushCounterService = async ( callback ) =>
{
    const value = await AsyncStorage.getItem( 'persist:ivo-app' );
    var itemcntr = 0;
    // console.log( '_________', value );
    if ( value != null )
    {
        const localStorage = JSON.parse( value ).PushCounterReducer;
        itemcntr = JSON.parse( localStorage ).PUSH_COUNTER_CHECK;
        console.log( 'PUSH_COUNTER reducer', itemcntr );

        return callback( itemcntr );
    } else
    {
        return callback( true );
    }
};
